function validateSpecialChar(){
	var obj=document.forms[0].elements;
	var len=obj.length;
	var funcName="this.custValidateSpecialChar";
	var isFuncAvl=(eval(funcName)!=undefined);
	var retVal=true;
	var invalidChars=/[\|\n]/g;
	for(var i=0;i<len;i++){
		switch(obj[i].type){
			case"text":case"textarea":
			var textval=obj[i].value;
			if(!obj[i].readOnly&&!obj[i].disabled){
				if((obj[i].id == "trmsCdns") || (obj[i].id == "remarks")){
					invalidChars=/[\|]/g;
				}else{
					invalidChars=/[\|\n]/g;
				}
				if(invalidChars.test(textval)){
					alert(finbranchResArr.get("FAT000485"));
					obj[i].focus();
					retVal=false;
					return retVal;
					break
				}
				if(isFuncAvl){
					if(eval(funcName).call(this,textval)==false){
						obj[i].focus();
						retVal=false;return retVal;
						break
					}
				}
			}
			break;
			default:break
		}
	}
	return retVal
}

function fnExecuteScriptForListWithHeader(headerNames,headerValues,inputNameValues, outputNames, scrName, pageTitle, literalNames, hyperLnkCols, isPopulationReq)
{

	if((inputNameValues.indexOf("pageNumber") != -1) && (inputNameValues.indexOf("pageSize") == -1))
	{
		alert("Page Size is mandatory");
		return;
	}
		
	if((inputNameValues.indexOf("pageSize") != -1) && (inputNameValues.indexOf("pageNumber") == -1))
	{
		alert("Page Number is mandatory");
		return;
	}

	if(!cust_fnIsNull(headerNames))
		sUrl += "&headerNames="+encodeURIComponent(headerNames);
		
	if(!cust_fnIsNull(headerValues))
		sUrl += "&headerValues="+encodeURIComponent(headerValues);
	
	if(!cust_fnIsNull(inputNameValues))
		sUrl += "&inputs="+encodeURIComponent(inputNameValues);


	if(cust_fnIsNull(scrName))
	{
		alert("Script Name is mandatory");
		return;
	}
	
	sUrl += "&scrName="+scrName;


	if(cust_fnIsNull(outputNames))
	{
		alert("List ouput field names are mandatory");
		return;
	}

	sUrl += "&outputs="+outputNames;

	if(cust_fnIsNull(pageTitle))
	{
		pageTitle=finbranchResource.FHP000368;
	}

	sUrl += "&pageTitle="+encodeURIComponent(pageTitle);

	if(cust_fnIsNull(literalNames))
	{
		alert("Column Literal Names are mandatory");
		return;
	}
	
	sUrl += "&literalNames="+encodeURIComponent(literalNames);	
	
	if(cust_fnIsNull(hyperLnkCols))
	{
		hyperLnkCols="1";
	}
	sUrl += "&hyperLnkCols="+hyperLnkCols;
	
	var retVal = cust_popModalWindow(sUrl, "Custom List");
	
	if (retVal == null || retVal == undefined)
		return retVal;
	
	if (!isPopulationReq)
		return retVal;
	
	if ("Microsoft Internet Explorer" == browser_name)
	{
		var frm = document.forms[0];
		var outBuff = outputNames.split("|");
		var outBuffLen = outBuff.length;
		var retBuff = retVal.split("|");			

		for (var i=0; i<outBuffLen; i++)

			if ((eval("frm." + outBuff[i]) != undefined))
			{
				eval("frm."+outBuff[i]+".value=\""+retBuff[i]+"\"");
			}
		}
	}
}

function fnConvertSqlStatement(sQry){
	sQry = sQry.toUpperCase();	
	/********************************************************************
	KEYWORD AND CHARACTER REPLACEMENT FOR SQL QUERY
	/********************************************************************
	( 			-> 			ABC#XYZ#OB
	)			->			ABC#XYZ#CB
	SELECT		->			ABC#XYZ#S
	FROM		->			ABC#XYZ#F
	WHERE		->			ABC#XYZ#W
	UNION		->			ABC#XYZ#U
	UNION ALL	->			ABC#XYZ#UA
	SUM			->			ABC#XYZ#M
	COUNT		->			ABC#XYZ#C
	GROUP		->			ABC#XYZ#G
	********************************************************************/
	
	sQry = sQry.replace(/\(/g,"ABC#XYZ#OB");
	sQry = sQry.replace(/\)/g,"ABC#XYZ#CB");
	
	sQry = sQry.replace(/SELECT/g,"ABC#XYZ#S");
	sQry = sQry.replace(/FROM/g,"ABC#XYZ#F");
	sQry = sQry.replace(/WHERE/g,"ABC#XYZ#W");
	sQry = sQry.replace(/UNION/g,"ABC#XYZ#U");
	sQry = sQry.replace(/UNION ALL/g,"ABC#XYZ#UA");
	sQry = sQry.replace(/SUM/g,"ABC#XYZ#M");
	sQry = sQry.replace(/COUNT/g,"ABC#XYZ#C");
	sQry = sQry.replace(/GROUP/g,"ABC#XYZ#G");
	sQry = sQry.replace(/'/g,"ABC#XYZ#Q");
	return sQry;
}

function fnCheckDuplicateRecord(obj,currentPos){
    ObjForm = document.forms[0];
    var hashMapArray = this["hashMapArray" + obj.id];
    var n = hashMapArray.length;
    var isDup = false;
    for(i=0;i<n;i++)
    {
        if((hashMapArray[i] === obj.value) && (i != currentPos)){
            alert(obj.value+' record already exists');
            obj.focus();
            obj.value = "";
            isDup = true;
            return (!isDup);
        }
    }
    return (!isDup);
}

    ObjForm = document.forms[0];
    var n = hashMapArray.length;
    var ObjectArray = fieldObjects.split("|");

    }
    var isDup = false;

                    document.getElementById(ObjectArray[j]).value = "";

        }

       return (!isDup);
}
